Animation & Interaction System

GSAP + ScrollTrigger Setup

// lib/animations.ts
export const scrollReveal = (element: Element, options = {}) => {
  gsap.from(element, {
    scrollTrigger: {
      trigger: element,
      start: "top 80%",
      toggleActions: "play none none reverse",
    },
    y: 60,
    opacity: 0,
    duration: 0.8,
    ease: "power3.out",
    ...options
  });
};

export const staggerChildren = (parent: Element, selector: string) => {
  gsap.from(parent.querySelectorAll(selector), {
    scrollTrigger: { trigger: parent, start: "top 85%" },
    y: 30,
    opacity: 0,
    duration: 0.6,
    stagger: 0.08,
    ease: "power2.out"
  });
};

R3F Animation Patterns

// hooks/useScrollParallax.ts
export const useScrollParallax = (factor = 0.5) => {
  const ref = useRef(null);
  const scrollY = useScroll(); // from @react-three/drei

  useFrame(() => {
    if (ref.current) {
      ref.current.position.y = -scrollY.current * factor;
    }
  });

  return ref;
};

Accessibility Checklist

  • prefers-reduced-motion: Disable all 3D animation, show static fallback
  • prefers-contrast: high: Increase border weights, disable glassmorphism
  • Keyboard navigation: All 3D interactions have keyboard equivalents
  • Screen readers: ARIA labels on Canvas, hidden text alternatives
  • Color contrast: WCAG AA minimum (4.5:1 text, 3:1 UI)
  • Focus indicators: Visible on all interactive 3D elements
  • Touch targets: Minimum 44x44px on mobile

GitHub Actions Workflow

# .github/workflows/deploy.yml
name: Deploy to Vercel
on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
      - run: npm ci
      - run: npm run lint
      - run: npm run build
      - uses: amondnet/vercel-action@v25
        if: github.ref == 'refs/heads/main'
        with:
          vercel-token: ${{ secrets.VERCEL_TOKEN }}
          vercel-org-id: ${{ secrets.VERCEL_ORG_ID }}
          vercel-project-id: ${{ secrets.VERCEL_PROJECT_ID }}
          vercel-args: '--prod'
